# STORE a LOAD späť, aj keď je dátový a adresový register ten istý
1210    ;  0  MVI  R2, 0x10        adresa
1334    ;  1  MVI  R3, 0x34
3312    ;  2  MVIB R3, 0x12       R3 = 0x1234
5320    ;  3  STORE R3, [R2]      mem[0x10] = 0x1234
4420    ;  4  LOAD  R4, [R2]      R4 = 0x1234
1520    ;  5  MVI  R5, 0x20
5550    ;  6  STORE R5, [R5]      mem[0x20] = 0x20 (Rs = Ra)
4650    ;  7  LOAD  R6, [R5]      R6 = 0x20
4220    ;  8  LOAD  R2, [R2]      R2 = mem[0x10] (cieľ = adresový register)
expect r2=0x1234 r3=0x1234 r4=0x1234 r5=0x20 r6=0x20 mem[0x10]=0x1234 mem[0x20]=0x20
