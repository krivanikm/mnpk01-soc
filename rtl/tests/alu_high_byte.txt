32FF    ; MVIB R2, 0xFF  -> R2 = 0xFF01 (horný bajt sa nesmie pripočítať)
1302    ; MVI  R3, 2
7234    ; ALU  R2, R3, R4 -> R4[7:0] = 3, R4 = 0xAB03
